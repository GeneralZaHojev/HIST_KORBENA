ASG_military_update_stats = {
	every_country = {
		limit = { 
			NOT = { tag = TUR  } 
			NOT = { tag = SWI }
		}
		set_variable = {
			ASG_military_score = THIS.num_of_military_factories
		}
		set_variable = {
			ASG_military_score_modifier = modifier@industrial_capacity_factory
		}
		set_variable = {
			ASG_military_score_max_efficiency = modifier@production_factory_max_efficiency_factor
		}
		add_to_variable = {
			ASG_military_score_modifier = 1
		}
		add_to_variable = {
			ASG_military_score_max_efficiency = 1
		}
		clamp_variable = {
			var = ASG_military_score_modifier
			min = 0
			max = 10000
		}
		multiply_variable = {
			ASG_military_score = ASG_military_score_modifier
		}
		multiply_variable = {
			ASG_military_score = ASG_military_score_max_efficiency
		}
		multiply_variable = {
			ASG_military_score = 4.5
		}
		add_to_array = {
			THIS.ASG_military_score_array = ASG_military_score
		}
		# log = "score_var = [?ASG_military_score] THIS = [?THIS.GetName] ROOT = [?ROOT.GetName]"
	}

	TUR = {
		set_variable = {
			ASG_military_score_ALI = SOV.ASG_military_score
		}
		add_to_variable = {
			ASG_military_score_ALI = USA.ASG_military_score
		}
		add_to_variable = {
			ASG_military_score_ALI = ENG.ASG_military_score
		}
		add_to_variable = {
			ASG_military_score_ALI = FRA.ASG_military_score
		}
		add_to_variable = {
			ASG_military_score_ALI = AST.ASG_military_score
		}
		add_to_variable = {
			ASG_military_score_ALI = CAN.ASG_military_score
		}
		add_to_variable = {
			ASG_military_score_ALI = RAJ.ASG_military_score
		}
		add_to_variable = {
			ASG_military_score_ALI = SAF.ASG_military_score
		}
		add_to_array = { THIS.ASG_military_score_array = ASG_military_score_ALI }
	}

	SWI = {
		set_variable = {
			ASG_military_score_AXI = GER.ASG_military_score
		}
		add_to_variable = {
			ASG_military_score_AXI = ITA.ASG_military_score
		}

ASG_military_rescale_up = {
	if = {
		limit = {
			check_variable = {
				GLOBAL.ASG_military_rendering_scale_factor > 0.125
			}
		}
		multiply_variable = {
			GLOBAL.ASG_military_rendering_scale_factor = 0.5
		}
		ASG_replot_military_graph = yes
	}
}
ASG_military_rescale_down = {
	if = {
		limit = {
			check_variable = {
				GLOBAL.ASG_military_rendering_scale_factor < 8
			}
		}
		multiply_variable = {
			GLOBAL.ASG_military_rendering_scale_factor = 2
		}
		ASG_replot_military_graph = yes
	}
}
